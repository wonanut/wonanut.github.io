/* ─────────────────────────────────────────────────────────────
 * token 估算
 * CJK 约 1 token/字，其他约 4 字符/token，向上取整；用码点遍历避免拆开 emoji 代理对。
 * ───────────────────────────────────────────────────────────── */
export function estimateTokens(text: unknown): number {
  if (text === null || text === undefined) return 0;
  const s = String(text);
  if (!s) return 0;
  let cjk = 0;
  let other = 0;
  for (const ch of s) {
    if (isCjkChar(ch)) cjk += 1;
    else other += 1;
  }
  return Math.ceil(cjk + other / 4);
}

function isCjkChar(ch: string): boolean {
  const code = ch.codePointAt(0) ?? 0;
  if (code >= 0x2e80 && code <= 0x9fff) return true; // CJK 部首、汉字、假名、CJK 标点
  if (code >= 0xf900 && code <= 0xfaff) return true; // CJK 兼容汉字
  if (code >= 0xff00 && code <= 0xffef) return true; // 全角字符与标点
  return false;
}

/** 最多 n 个字符（含省略号），按码点裁剪。 */
export function truncate(text: unknown, n: number): string {
  const s = text === null || text === undefined ? '' : String(text);
  if (typeof n !== 'number' || n <= 0) return '';
  const chars = Array.from(s);
  if (chars.length <= n) return chars.join('');
  if (n <= 1) return '…';
  return chars.slice(0, n - 1).join('') + '…';
}

/** 折叠空白并裁剪到最多 max 个字符。 */
export function cleanText(text: unknown, max?: number): string {
  const s = (text === null || text === undefined ? '' : String(text)).replace(/\s+/g, ' ').trim();
  if (typeof max === 'number' && max > 0) return truncate(s, max);
  return s;
}

/** 按 token 预算裁剪文本：保证 estimateTokens(结果) <= maxTokens。 */
export function truncateToTokens(text: unknown, maxTokens: number): string {
  const s = text === null || text === undefined ? '' : String(text);
  if (typeof maxTokens !== 'number' || maxTokens <= 0) return '';
  if (estimateTokens(s) <= maxTokens) return s;
  const chars = Array.from(s);
  let lo = 0;
  let hi = chars.length;
  while (lo < hi) {
    const mid = Math.ceil((lo + hi) / 2);
    const candidate = chars.slice(0, mid).join('') + '…';
    if (estimateTokens(candidate) <= maxTokens) lo = mid;
    else hi = mid - 1;
  }
  return lo > 0 ? chars.slice(0, lo).join('') + '…' : '';
}

export function clone<T>(value: T): T {
  if (value === null || typeof value !== 'object') return value;
  if (Array.isArray(value)) return value.map((v) => clone(v)) as unknown as T;
  const out: Record<string, unknown> = {};
  for (const key of Object.keys(value as Record<string, unknown>)) {
    out[key] = clone((value as Record<string, unknown>)[key]);

  return out as unknown as T;
}

/* ── 安全计算器：词法分析 + 递归下降，不 eval ── */
type CalcToken =
  | { type: 'num'; value: number }
  | { type: '+' | '-' | '*' | '/' | '%' | '^' | '(' | ')' };

function calcTokenize(expression: string): CalcToken[] | null {
  const source = String(expression).replace(/\s+/g, '');
  if (!source) return null;
  const tokens: CalcToken[] = [];
  let i = 0;
  while (i < source.length) {
    const ch = source.charAt(i);
    if (ch >= '0' && ch <= '9') {
      let j = i;
      let dots = 0;
      while (j < source.length && (source.charAt(j) === '.' || (source.charAt(j) >= '0' && source.charAt(j) <= '9'))) {
        if (source.charAt(j) === '.') dots += 1;
        j += 1;
      }
      if (dots > 1) return null;
      const num = Number(source.slice(i, j));
      if (!isFinite(num)) return null;
      tokens.push({ type: 'num', value: num });
      i = j;
      continue;
    }
    if ('+-*/%^()'.indexOf(ch) >= 0) {
      tokens.push({ type: ch as '+' | '-' | '*' | '/' | '%' | '^' | '(' | ')' });
      i += 1;
      continue;
    }
    return null; // 出现不认识的字符
  }
  return tokens;
}

function calcParse(tokens: CalcToken[]): number {
  let pos = 0;
  const peek = (): CalcToken | undefined => tokens[pos];
  function fail(message: string): never {
    throw new Error(message);
  }

  function parseExpression(): number {
    let left = parseTerm();
    while (peek() && (peek()!.type === '+' || peek()!.type === '-')) {
      const op = tokens[pos].type as string;
      pos += 1;
      const right = parseTerm();
      left = op === '+' ? left + right : left - right;
    }
    return left;
  }
  function parseTerm(): number {
    let left = parseUnary();
    while (peek() && (peek()!.type === '*' || peek()!.type === '/' || peek()!.type === '%')) {
      const op = tokens[pos].type as string;
      pos += 1;
      const right = parseUnary();
      if (op === '*') left = left * right;
      else if (op === '/') {
        if (right === 0) fail('除数不能为 0');
        left = left / right;
      } else {
        if (right === 0) fail('取余的除数不能为 0');
        left = left % right;
      }
    }
    return left;
  }
  function parseUnary(): number {
    if (peek() && peek()!.type === '-') {
      pos += 1;
      return -parseUnary();
    }
    if (peek() && peek()!.type === '+') {
      pos += 1;
      return parseUnary();
    }
    return parsePower();
  }
  function parsePower(): number {
    const base = parsePrimary();
    if (peek() && peek()!.type === '^') {
      pos += 1;
      return Math.pow(base, parseUnary()); // 右结合，允许 2^-1
    }
    return base;
  }
  function parsePrimary(): number {
    const token = peek();
    if (!token) fail('表达式不完整');
    if (token.type === 'num') {
      pos += 1;
      return token.value;
    }
    if (token.type === '(') {
      pos += 1;
      const value = parseExpression();
      if (!peek() || peek()!.type !== ')') fail('括号不匹配');
      pos += 1;
      return value;
    }
    fail('这里需要一个数字或左括号');
    return 0;
  }

  const result = parseExpression();
  if (pos !== tokens.length) fail('表达式有多余内容');
  return result;
}

}

  parameters: {
    type: 'object',
    properties: { expression: { type: 'string', description: '要计算的表达式，例如 (12+30)*3' } },

  },
  run(args) {

      const expression = typeof args?.expression === 'string' ? args.expression : '';
      if (!expression.trim()) return toolError('缺少表达式');

/* 从「我的X是什么」里抽关键词；泛化问法返回空串（让 memory_search 返回最近记忆）。 */
function recallQuery(input: string): string {
  if (/^(你还记得什么|你还记得些什么|还记得什么|你记得什么|记得什么|记忆|回忆|回忆一下|你还记得吗|你记得吗)[？?。！!\s]*$/.test(input)) {
    return '';
  }
  const matched = /我的(.{1,24}?)(?:是什么|是多少|呢|叫什么|叫啥|多少)/.exec(input);
  if (matched && matched[1].trim()) return matched[1].trim();
  const stripped = input
    .replace(/你还记得|还记不记得|记得|回忆|记忆|一下|帮我|请问|吗|呢|什么|查|找|的|是/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
  return stripped || input;
}

/* 从输入里抽算式；抽不到返回 null。 */
function extractExpression(input: string): string | null {
  const cleaned = input.replace(/[^0-9+\-*/%^().\s]/g, ' ').replace(/\s+/g, '');
  if (!cleaned) return null;
  if (!/[0-9]/.test(cleaned)) return null;
  if (!/[+\-*/%^]/.test(cleaned)) return null;
  let depth = 0;
  for (const ch of cleaned) {
    if (ch === '(') depth += 1;
    else if (ch === ')') {
      depth -= 1;
      if (depth < 0) return null;
    }
  }
  if (depth !== 0) return null;
  if (/[+\-*/%^]$/.test(cleaned)) return null;
  if (/^[*/%^]/.test(cleaned)) return null;
  return cleaned;
}

function trimPunct(text: unknown): string {
  return String(text == null ? '' : text)
    .replace(/^[「『"'“\s]+|[」』"'”\s]+$/g, '')
    .replace(/[。！!？?；;，,、\s]+$/g, '')
    .trim();
}

/* 解析「记住 …」→ { key, value }；失败返回 null。 */
function extractMemory(input: string): { key: string; value: string } | null {
  let text = input;
  const stripped = text.replace(
    /^(请|麻烦|帮我|帮忙)?\s*(记住|记一下|记一记|记下|记好|保存一下|保存|存一下|存下)[：:，,、\s]*/,
    '',
  );
  if (stripped !== text) text = stripped.trim();
  if (!text) return null;
  let m = /^我的(.{1,20}?)是(.+)$/.exec(text);
  if (m) return { key: trimPunct(m[1]), value: trimPunct(m[2]) };
  m = /^我的(.{1,20}?)叫(.+)$/.exec(text);
  if (m) return { key: trimPunct(m[1]), value: trimPunct(m[2]) };
  m = /^(.{1,20}?)\s*是\s*(.+)$/.exec(text);
  if (m) return { key: trimPunct(m[1]), value: trimPunct(m[2]) };
  m = /^(.{1,30}?)\s*(?:=|：|:)\s*(.+)$/.exec(text);
  if (m) return { key: trimPunct(m[1]), value: trimPunct(m[2]) };
  return { key: '备忘', value: trimPunct(text) };
}

export function nextStep(ctx: LlmContext): ModelDecision {

  if (ctx?.lastTool && ctx.lastTool.name) return continueAfterTool(ctx, input);

  if (!input) return { type: 'final', text: EMPTY_TEXT };
  if (isHelpQ(input)) return { type: 'final', text: HELP_TEXT };

  if (isToolsQ(input)) return { type: 'final', text: TOOLS_TEXT };
  if (isPrincipleQ(input)) return { type: 'final', text: PRINCIPLE_TEXT };
  if (isForgetQ(input)) return { type: 'final', text: FORGET_TEXT };
  if (isCompactQ(input)) return { type: 'final', text: compactText(ctx) };

  if (isTimeQ(input)) return { type: 'tool_call', name: 'clock', args: { format: 'full' } };

  const expression = extractExpression(input);
  if (expression) return { type: 'tool_call', name: 'calculator', args: { expression } };

  if (isRecallQ(input)) return { type: 'tool_call', name: 'memory_search', args: { query: recallQuery(input) } };

  if (isWriteQ(input)) {
    const memory = extractMemory(input);
    if (!memory || !memory.key || !memory.value) {

    }
    return { type: 'tool_call', name: 'memory_write', args: { key: memory.key, value: memory.value } };
  }

  return { type: 'final', text: fallbackText(ctx, input) };
}

function continueAfterTool(ctx: LlmContext, input: string): ModelDecision {

    const expression = args.expression ? String(args.expression) : '';
    return { type: 'final', text: (expression ? expression + ' = ' + display + '。' : display + '。') + '这是 calculator 工具真实算出来的结果，我只负责决定调用它。' };
  }

  if (tool.name === 'clock') {
    if (!tool.ok) return { type: 'final', text: '读取时间失败了：' + (reason || '未知原因') + '。' };
    if (save) return { type: 'tool_call', name: 'memory_write', args: { key: '当前时间', value: display } };
    const weekday = result.weekday ? '（' + result.weekday + '）' : '';
    return { type: 'final', text: '现在是 ' + display + weekday + '。这是 clock 工具返回的本机时间。' };
  }

  if (tool.name === 'memory_write') {
    if (!tool.ok) return { type: 'final', text: '保存记忆失败了：' + (reason || '未知原因') + '。' };
    const key = result.key !== undefined ? String(result.key) : String(args.key || '');
    const value = result.value !== undefined ? String(result.value) : String(args.value || '');
    return { type: 'final', text: '已经记住：' + key + ' = ' + value + '。之后你问「' + key + '是什么」，我会用 memory_search 检索出来。' };
  }

  if (tool.name === 'memory_search') {
    if (!tool.ok) return { type: 'final', text: '检索记忆失败了：' + (reason || '未知原因') + '。' };
    const matches = Array.isArray(result.matches) ? (result.matches as Array<{ key: string; value: string }>) : [];
    if (!matches.length) return { type: 'final', text: '我这边没有找到相关记忆——你还没告诉过我，或者它已经被清空了。要不要现在「记住 …」一下？' };
    const facts = matches.slice(0, 3).map((item) => item.key + ' = ' + item.value).join('；');
    return { type: 'final', text: '我记得：' + facts + '。这些是从长期记忆里检索到的，跨会话共享。' };
  }

  return { type: 'final', text: '工具 ' + tool.name + ' 返回：' + (display || '（空结果）') + '。' };
}

export interface StreamOptions {
  pace: Pace;
  sleep: (ms: number) => Promise<void>;
  cancelled: () => boolean;
  onToken: (chunk: string, isLast: boolean) => void;
}

export async function streamAnswer(text: string, opts: StreamOptions): Promise<boolean> {
  const { sleep, cancelled, onToken, pace } = opts;
  const base = LAB_PACE_DELAYS[normalizePace(pace)] ?? LAB_PACE_DELAYS.normal;
  const chars = Array.from(String(text ?? ''));
  if (!chars.length) return true;

  const chunkSize = Math.max(1, Math.ceil(chars.length / 60));
  const chunks: string[] = [];
  for (let i = 0; i < chars.length; i += chunkSize) chunks.push(chars.slice(i, i + chunkSize).join(''));
  const interval = Math.max(1, Math.min(base, Math.floor(1500 / Math.max(1, chunks.length))));

  for (let i = 0; i < chunks.length; i += 1) {
    if (cancelled()) return false;
    await sleep(interval);
    if (cancelled()) return false;
    onToken(chunks[i], i === chunks.length - 1);
  }
  return true;
}

      existing.value = input.value;
      existing.source = input.source;
      existing.sessionId = input.sessionId;
      existing.updatedAt = now();
      return existing;
    }
    const item: MemoryItem = {
      id: id('mem'),
      key: input.key,
      value: input.value,
      source: input.source,
      sessionId: input.sessionId,
      createdAt: now(),
      updatedAt: now(),
    };
    memory.push(item);
    if (memory.length > LAB_MEMORY_LIMIT) memory = memory.slice(-LAB_MEMORY_LIMIT);
    return item;
  }

  function searchMemory(query: string): MemoryItem[] {
    const q = cleanText(query).toLowerCase();
    if (!q) return memory.slice(-3).reverse();
    const scored = memory
      .map((m) => {
        const key = m.key.toLowerCase();
        const value = m.value.toLowerCase();
        let score = 0;
        if (key.includes(q)) score += 3;
        if (value.includes(q)) score += 2;
        if (q.includes(key) || key.includes(q)) score += 1;
        return { m, score };
      })
      .filter((x) => x.score > 0)
      .sort((a, b) => b.score - a.score || b.m.updatedAt - a.m.updatedAt);
    return scored.slice(0, 3).map((x) => x.m);
  }

  function recallMemory(input: string): MemoryItem[] {
    return searchMemory(input);
  }

  /* ── 上下文组装 ── */

  function assembleContext(session: SessionData, input: string, excludeId?: string): ContextView {
    const segments: ContextView['segments'] = [];
    const recent = session.messages.slice(session.foldedCount).filter((m) => m.id !== excludeId);
    const hits = recallMemory(input);
    let used = 0;

    const sysTokens = estimateTokens(LAB_SYSTEM_PROMPT);
    segments.push({ id: 'system', label: '系统提示', tokens: sysTokens });
    used += sysTokens;

    if (session.summary) {
      const t = estimateTokens(session.summary);
      segments.push({ id: 'summary', label: '历史摘要', tokens: t });
      used += t;
    }

    if (hits.length) {
      const t = estimateTokens(hits.map((h) => h.key + ' = ' + h.value).join('\n'));
      segments.push({ id: 'memory', label: '记忆召回', tokens: t });
      used += t;
    }

    const recentTokens = recent.reduce((s, m) => s + m.tokens + LAB_MSG_OVERHEAD, 0);
    segments.push({ id: 'recent', label: '最近消息', tokens: recentTokens });
    used += recentTokens;

    const inputTokens = estimateTokens(input);
    segments.push({ id: 'input', label: '本轮输入', tokens: inputTokens });
    used += inputTokens;

    return { limit: budget, used, segments, hits, summary: session.summary, messageCount: session.messages.length };
  }

  function assemblePromptText(session: SessionData, input: string): string {
    const parts: string[] = ['[system] ' + LAB_SYSTEM_PROMPT];
    if (session.summary) parts.push('[summary] ' + session.summary);
    const hits = recallMemory(input);
    if (hits.length) parts.push('[memory] ' + hits.map((h) => h.key + ' = ' + h.value).join('\n'));
    for (const m of session.messages.slice(session.foldedCount)) {
      if (m.id === inputMsgId) continue;
      parts.push('[' + m.role + '] ' + m.content);
    }
    parts.push('[input] ' + input);
    return parts.join('\n');
  }

  function buildRealMessages(session: SessionData, input: string): OpenAiMessage[] {
    const systemParts = [LAB_SYSTEM_PROMPT];
    if (session.summary) systemParts.push('【历史摘要】\n' + session.summary);
    const hits = recallMemory(input);
    if (hits.length) systemParts.push('【长期记忆】\n' + hits.map((h) => h.key + ' = ' + h.value).join('\n'));

    const msgs: OpenAiMessage[] = [{ role: 'system', content: systemParts.join('\n\n') }];
    for (const m of session.messages.slice(session.foldedCount)) {
      if (m.id === inputMsgId) continue;
      if (m.role === 'tool') msgs.push({ role: 'user', content: '[工具结果: ' + (m.toolName || '') + '] ' + m.content });
      else msgs.push({ role: m.role === 'assistant' ? 'assistant' : 'user', content: m.content });
    }
    msgs.push({ role: 'user', content: input });
    return msgs;
  }

  /* ── 压缩 ── */

  function summarizeMessages(messages: Message[]): string {
    const lines: string[] = [];
    for (const m of messages) {
      const first = firstSentence(m.content);
      let s = '';
      if (m.role === 'user') s = '问：' + first;
      else if (m.role === 'tool') s = '工具 ' + (m.toolName || '') + ' 返回：' + first;
      else if (m.role === 'assistant') s = '答：' + first;
      if (s) lines.push(truncate(s, 60));
    }
    return lines.slice(-LAB_MAX_SUMMARY_LINES).join('\n');
  }

  function firstSentence(text: string): string {
    const end = text.search(/[。\n]/);
    return (end === -1 ? text : text.slice(0, end + 1)).replace(/\s+/g, ' ').trim();
  }

  function compactSession(session: SessionData): { before: number; after: number; folded: number } | null {
    const live = session.messages.slice(session.foldedCount);
    if (live.length <= recentKeep) return null;
    const foldCount = live.length - recentKeep;
    const toFold = session.messages.slice(0, session.foldedCount + foldCount);
    const before = assembleContext(session, lastInput, inputMsgId).used;
    const newSummary = summarizeMessages(toFold);
    const maxSummary = Math.max(1, Math.floor(budget * LAB_SUMMARY_TOKEN_RATIO));
    session.summary = session.summary
      ? truncateToTokens(session.summary + '\n' + newSummary, maxSummary)
      : truncateToTokens(newSummary, maxSummary);
    session.foldedCount += foldCount;
    session.compressions += 1;
    stats.compressions += 1;
    const after = assembleContext(session, lastInput, inputMsgId).used;
    const saved = before - after;
    if (saved > 0) stats.tokensSaved += saved;
    return { before, after, folded: foldCount };
  }

  /* ── turn 循环 ── */

  function addStage(turn: Turn, kind: Stage['kind'], title: string, meta: string, detail?: unknown): Stage {
    const stage: Stage = { id: id('st'), kind, title, meta, status: 'active', detail: detail ?? null, at: now() };
    turn.stages.push(stage);
    emit({ type: 'stage' });
    return stage;
  }

  async function runTool(name: string, args: Record<string, unknown>, session: SessionData): Promise<ToolResult> {
    const tool = LAB_TOOLS.find((t) => t.name === name);
    if (!tool) return { ok: false, result: { error: '未找到工具 ' + name }, display: '错误：未找到工具 ' + name };
    const toolCtx: ToolContext = {
      now,
      sessionId: session.id,
      writeMemory,
      searchMemory,
    };
    try {
      return await tool.run(args, toolCtx);
    } catch (err) {
      const msg = errorMessage(err);
      return { ok: false, result: { error: msg }, display: '错误：' + msg };
    }
  }

  async function streamOut(session: SessionData, text: string, stage: Stage): Promise<void> {
    const line: TranscriptLine = { id: id('ln'), kind: 'agent', text: '', at: now(), streaming: true };
    session.transcript.push(line);
    if (session.transcript.length > LAB_MAX_TRANSCRIPT) session.transcript = session.transcript.slice(-LAB_MAX_TRANSCRIPT);
    emit({ type: 'line' });
    streamingLine = { lineId: line.id, text: '' };

    await streamAnswer(text, {
      pace,
      sleep,
      cancelled: () => cancelled || disposed,
      onToken: (chunk, isLast) => {
        line.text += chunk;
        line.streaming = !isLast;
        streamingLine = { lineId: line.id, text: line.text };
        emit({ type: 'token', text: chunk });
      },
    });

    line.streaming = false;
    streamingLine = null;
    const msg = newMessage('assistant', text, 'final');
    pushMessage(session, msg);
    stage.status = 'done';
    emit({ type: 'message' });
    emit({ type: 'line' });
    emit({ type: 'stage' });
  }

  async function submit(text: string): Promise<void> {
    if (busy || disposed) return;
    const input = cleanText(text, LAB_MAX_INPUT_CHARS);
    if (!input) return;

    busy = true;
    cancelled = false;
    lastInput = input;
    const session = activeSession();
    if (!session) {
      busy = false;
      return;
    }

    session.turns += 1;
    stats.turns += 1;
    const turnId = id('turn');
    const turn: Turn = { id: turnId, input, status: 'running', startedAt: now(), endedAt: null, stages: [] };
    currentTurn = turn;
    emit({ type: 'turn_start', turnId });

    try {
      // 输入
      const inputMsg = newMessage('user', input);
      inputMsgId = inputMsg.id;
      pushMessage(session, inputMsg);
      pushLine(session, 'user', input);
      addStage(turn, 'input', '输入', charCount(input) + ' 字 · ' + estimateTokens(input) + ' tok');
      emit({ type: 'message' });
      emit({ type: 'line' });

      // 组装上下文（压缩前）
      let ctx = assembleContext(session, input, inputMsgId);
      contextView = ctx;
      addStage(turn, 'context', '组装上下文', ctx.used + ' / ' + budget + ' tok', {
        segments: ctx.segments,
        hits: ctx.hits.map((h) => h.key + '=' + h.value),
      });
      emit({ type: 'context' });

      // 压缩检查
      if (ctx.used > budget * LAB_COMPRESS_RATIO) {
        const r = compactSession(session);
        if (r) {
          addStage(turn, 'compress', '压缩上下文', r.before + ' → ' + r.after + ' tok', r);
          pushLine(session, 'compress', '⇣ 上下文已压缩 · ' + r.before + ' → ' + r.after + ' tok');
          emit({ type: 'compression' });
          emit({ type: 'line' });
          ctx = assembleContext(session, input, inputMsgId);
          contextView = ctx;
          emit({ type: 'context' });
        }
      }

      // 模型循环
      let finalText = '';
      let step = 0;
      let lastTool: LlmLastTool | null = null;
      let realMessages: OpenAiMessage[] | null = null;

      while (step < LAB_MAX_STEPS) {
        step += 1;
        if (cancelled || disposed) break;

        let decision: ModelDecision;
        let promptDetail: unknown;
        if (realClient) {
          realMessages = realMessages ?? buildRealMessages(session, input);
          decision = await realClient.complete({ messages: realMessages, tools: LAB_TOOL_SCHEMAS });
          promptDetail = realMessages;
        } else {
          decision = nextStep({ input, step, lastTool, hits: contextView.hits, usage: { used: contextView.used, limit: budget } });
          promptDetail = assemblePromptText(session, input);
        }
        const modelStage = addStage(turn, 'model', '模型决策 · 第 ' + step + ' 步', '', {
          step,
          prompt: promptDetail,
          response: decision,
        });
        modelStage.status = 'done';

        if (decision.type === 'tool_call') {
          const toolStage = addStage(turn, 'tool', '工具调用 · ' + decision.name, json(decision.args));
          pushLine(session, 'tool', '⚙ ' + decision.name + ' ' + json(decision.args));
          emit({ type: 'tool_call', name: decision.name });
          emit({ type: 'line' });

          const res = await runTool(decision.name, decision.args, session);
          toolStage.meta = res.ok ? '返回 ' + res.display : res.display;
          toolStage.status = res.ok ? 'done' : 'error';
          toolStage.detail = { name: decision.name, args: decision.args, result: res.result, display: res.display, ok: res.ok };
          emit({ type: 'stage' });

          pushMessage(session, newMessage('tool', res.display, 'tool_result', decision.name));
          pushLine(session, 'toolresult', '← ' + res.display);
          emit({ type: 'tool_result', name: decision.name, ok: res.ok });
          emit({ type: 'message' });
          emit({ type: 'line' });

          if (decision.name === 'memory_write' && res.ok) {
            const r = res.result as { key?: string; value?: string };
            pushLine(session, 'memory', '◈ 记忆已保存 · ' + (r.key || '') + ' = ' + (r.value || ''));
            emit({ type: 'memory' });
            emit({ type: 'line' });
          }

          stats.toolCalls += 1;
          lastTool = { name: decision.name, ok: res.ok, args: decision.args, result: res.result, display: res.display };
          if (realClient && realMessages) realMessages = appendToolExchange(realMessages, decision, res.display);

          ctx = assembleContext(session, input, inputMsgId);
          contextView = ctx;
          emit({ type: 'context' });
          continue;
        }

        finalText = decision.text;
        break;
      }

      if (!finalText) throw new Error('超过 ' + LAB_MAX_STEPS + ' 步仍未收敛，可能是工具循环没终止');

      // 输出
      const outStage = addStage(turn, 'output', '输出', estimateTokens(finalText) + ' tok', { text: finalText });
      await streamOut(session, finalText, outStage);
      turn.status = 'done';
      turn.endedAt = now();
    } catch (err) {
      turn.status = 'error';
      turn.endedAt = now();
      const msg = errorMessage(err);
      addStage(turn, 'output', '错误', msg).status = 'error';
      pushLine(session, 'error', '✗ ' + msg);
      emit({ type: 'error', message: msg });
      emit({ type: 'line' });
    } finally {
      busy = false;
      currentTurn = null;
      streamingLine = null;
      persist();
      emit({ type: 'turn_end', turnId });
      emit({ type: 'stage' });
    }
  }

  /* ── 会话 / 记忆操作 ── */

  function newSessionInternal(): SessionData {
    const s: SessionData = {
      id: id('sess'),
      title: '新会话',
      createdAt: now(),
      updatedAt: now(),
      messageCount: 0,
      tokens: 0,
      compressions: 0,
      turns: 0,
      messages: [],
      transcript: [],
      summary: null,
      foldedCount: 0,
    };
    s.transcript.push({ id: id('ln'), kind: 'banner', text: LAB_BANNER, at: now() });
    sessions.unshift(s);
    trimSessions();
    activeId = s.id;
    contextView = { limit: budget, used: 0, segments: [], hits: [], summary: null, messageCount: 0 };
    persist();
    emit({ type: 'state' });
    return s;
  }

  function metaOf(s: SessionData): SessionMeta {
    return {
      id: s.id,
      title: s.title,
      createdAt: s.createdAt,
      updatedAt: s.updatedAt,
      messageCount: s.messageCount,
      tokens: s.tokens,
      compressions: s.compressions,
      turns: s.turns,
    };
  }

  function persist(): void {
    if (disposed) return;
    try {
      storage.setItem(
        LAB_STORAGE_KEY,
        json({
          sessions: sessions.map((s) => ({
            ...s,
            messages: s.messages.slice(-LAB_MAX_MESSAGES),
            transcript: s.transcript.slice(-LAB_MAX_TRANSCRIPT),
          })),
          memory,
          stats,
        }),
      );
    } catch {
      /* 存储失败不影响运行 */
    }
  }

  function load(): void {
    try {
      const raw = storage.getItem(LAB_STORAGE_KEY);
      if (!raw) return;
      const data = JSON.parse(raw) as { sessions?: SessionData[]; memory?: MemoryItem[]; stats?: LabStats };
      if (data && Array.isArray(data.sessions)) sessions = data.sessions;
      if (data && Array.isArray(data.memory)) memory = data.memory;
      if (data && data.stats) stats = data.stats;
      if (sessions.length) activeId = sessions[0].id;
    } catch {
      /* 解析失败按全新状态启动 */
    }
  }

  /* ── 对外 API ── */

  function snapshot(): Snapshot {
    const s = activeSession();
    return clone({
      busy,
      pace,
      activeId,
      sessions: sessions.map(metaOf),
      messages: s ? s.messages.slice(-LAB_MAX_MESSAGES) : [],
      transcript: s ? s.transcript.slice(-LAB_MAX_TRANSCRIPT) : [],
      memory: memory.slice(),
      context: contextView,
      turn: currentTurn ? clone(currentTurn) : null,
      streaming: streamingLine ? { ...streamingLine } : null,
      stats: { ...stats },
      lastInput,
      realModel: !!realClient,
    });
  }

  function compact(): void {
    if (busy || disposed) return;
    const session = activeSession();
    if (!session) return;
    const live = session.messages.slice(session.foldedCount);
    if (live.length <= recentKeep) {
      pushLine(session, 'info', '没有足够的历史可压缩（保留最近 ' + recentKeep + ' 条）');
      emit({ type: 'line' });
      persist();
      return;
    }
    const r = compactSession(session);
    if (r) {
      pushLine(session, 'compress', '⇣ 上下文已压缩 · ' + r.before + ' → ' + r.after + ' tok');
      contextView = assembleContext(session, lastInput, inputMsgId);
      emit({ type: 'compression' });
      emit({ type: 'line' });
      emit({ type: 'context' });
    }
    persist();
  }

  function newSession(): boolean {
    if (busy || disposed) return false;
    newSessionInternal();
    return true;
  }

  function switchSession(targetId: string): boolean {
    if (busy || disposed) return false;
    const target = sessions.find((s) => s.id === targetId);
    if (!target) return false;
    activeId = target.id;
    emit({ type: 'state' });
    return true;
  }

  function deleteSession(targetId: string): void {
    if (busy || disposed) return;
    sessions = sessions.filter((s) => s.id !== targetId);
    if (sessions.length === 0) {
      newSessionInternal();
    } else if (activeId === targetId) {
      activeId = sessions[0].id;
    }
    persist();
    emit({ type: 'state' });
  }

  function clearMemory(): void {
    memory = [];
    persist();
    emit({ type: 'memory' });
  }

  function deleteMemory(targetId: string): void {
    memory = memory.filter((m) => m.id !== targetId);
    persist();
    emit({ type: 'memory' });
  }

  function resetAll(): void {
    sessions = [];
    memory = [];
    stats = EMPTY_STATS();
    activeId = '';
    newSessionInternal();
    persist();
    emit({ type: 'state' });
  }

  function setPace(p: Pace): void {
    pace = normalizePace(p);
    emit({ type: 'state' });
  }

  function setRealClient(client: RealClient | null): void {
    realClient = client;
    emit({ type: 'state' });
  }

  function dispose(): void {
    disposed = true;
    cancelled = true;
    listeners.clear();
  }

  function onEvent(fn: (e: LabEvent) => void): () => void {
    listeners.add(fn);
    return () => listeners.delete(fn);
  }

  // 启动：恢复或新建
  load();
  if (sessions.length === 0) {
    newSessionInternal();
  } else {
    activeId = sessions[0].id;
  }

  return {
    snapshot,
    onEvent,
    submit,
    compact,
    newSession,
    switchSession,
    deleteSession,
    clearMemory,
    deleteMemory,
    resetAll,
    setPace,
    setRealClient,
    dispose,
  };
